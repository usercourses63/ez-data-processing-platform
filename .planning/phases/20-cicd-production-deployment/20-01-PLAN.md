Extend the existing ci.yml (do NOT replace it). Make these specific changes:

1. Add tag trigger to the `on` section:
   ```yaml
   on:
     push:
       branches: [ main ]
       tags: [ 'v*' ]
     pull_request:
       branches: [ main ]
   ```

2. Update the `docker` job's version calculation step to prefer git tag:
   ```yaml
   - name: Calculate version
     id: version
     run: |
       if [[ "${{ github.ref }}" == refs/tags/v* ]]; then
         VERSION="${GITHUB_REF#refs/tags/}"
       else
         VERSION="v${{ env.SEMANTIC_VERSION }}.${{ github.run_number }}"
       fi
       echo "VERSION=$VERSION" >> $GITHUB_OUTPUT
       echo "COMMIT_SHA=${{ github.sha }}" >> $GITHUB_OUTPUT
       echo "SHORT_SHA=$(echo ${{ github.sha }} | cut -c1-7)" >> $GITHUB_STEP_SUMMARY
   ```

3. Update the `docker` job's `if` condition to also run on tags:
   Change: `if: github.ref == 'refs/heads/main'`
   To: `if: github.ref == 'refs/heads/main' || startsWith(github.ref, 'refs/tags/v')`

4. Add a new `helm-validate` job after the `docker` job:
   ```yaml
   helm-validate:
     needs: docker
     if: github.ref == 'refs/heads/main' || startsWith(github.ref, 'refs/tags/v')
     runs-on: ubuntu-latest
     steps:
       - uses: actions/checkout@v4

       - name: Install Helm
         uses: azure/setup-helm@v4

       - name: Lint Helm chart
         run: helm lint helm/ez-platform/

       - name: Template validation (default values)
         run: |
           helm template ez-platform helm/ez-platform/ \
             --values helm/ez-platform/values.yaml \
             > /dev/null

       - name: Template validation (local values)
         run: |
           helm template ez-platform helm/ez-platform/ \
             --values helm/ez-platform/values.yaml \
             --values helm/ez-platform/values-local.yaml \
             > /dev/null
         continue-on-error: true
   ```
   The `continue-on-error: true` on local values is because values-local.yaml may not exist yet when this plan runs before plan 02.

Keep all existing jobs, steps, and structure intact. Only ADD the tag trigger, update version extraction, update docker if-condition, and add the helm-validate job.
