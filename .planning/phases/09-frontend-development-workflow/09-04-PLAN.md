```typescript
/**
 * Docs URL utility for environment-aware documentation links.
 *
 * In development (localhost/127.0.0.1): Links to standalone Docusaurus at localhost:3000/docs
 * In production: Links to /docs path on same origin (served by K8s Ingress/OCP Route)
 *
 * Per user decision: Runtime detection (not build-time env vars) for single build artifact.
 */

/**
 * Determines if the app is running in development mode.
 * Development is identified by localhost or 127.0.0.1 hostname.
 */
export const isDevelopment = (): boolean => {
  const hostname = window.location.hostname;
  return hostname === 'localhost' || hostname === '127.0.0.1';
};

/**
 * Gets the base URL for documentation portal.
 *
 * @returns Base URL for docs (e.g., "http://localhost:3000/docs" or "/docs")
 */
export const getDocsBaseUrl = (): string => {
  if (isDevelopment()) {
    // Development: Docusaurus runs separately on port 3000
    return 'http://localhost:3000/docs';
  }
  // Production: Docs served at /docs path via Ingress/Route
  return '/docs';
};

/**
 * Gets the full URL for a specific documentation page.
 *
 * @param path - Optional path within docs (e.g., "components/datasource-forms")
 * @returns Full URL to the documentation page
 *
 * @example
 * getDocsUrl() // Returns base docs URL
 * getDocsUrl('components/datasource-forms') // Returns URL to specific page
 * getDocsUrl('user-guide-he') // Returns URL to Hebrew user guide
 */
export const getDocsUrl = (path?: string): string => {
  const baseUrl = getDocsBaseUrl();
  if (!path) {
    return baseUrl;
  }
  // Ensure no double slashes
  const cleanPath = path.startsWith('/') ? path.slice(1) : path;
  return `${baseUrl}/${cleanPath}`;
};

/**
 * Opens the documentation portal in a new tab.
 *
 * @param path - Optional path within docs
 */
export const openDocs = (path?: string): void => {
  const url = getDocsUrl(path);
  window.open(url, '_blank', 'noopener,noreferrer');
};
```

```typescript
    await page.goto('/');

  test('should have Help button visible in header', async ({ page }) => {
    // Help button should be visible (QuestionCircleOutlined icon)

      has: page.locator('[aria-label*="question"], .anticon-question-circle')

    // Alternative selector if icon-based fails
    const helpByTooltip = page.getByRole('button', { name: /עזרה|help/i });

                      await helpByTooltip.isVisible().catch(() => false);

    expect(isVisible).toBe(true);
  });

  test('should open docs in new tab when Help clicked', async ({ page, context }) => {
    // Listen for new page/tab opening
    const pagePromise = context.waitForEvent('page');

    // Find and click Help button
    const helpButton = page.getByRole('button', { name: /עזרה|help/i });

    if (await helpButton.isVisible()) {
      await helpButton.click();

      // Wait for new tab to open
      const newPage = await pagePromise;
      await newPage.waitForLoadState('domcontentloaded');

      // Verify URL contains /docs
      const url = newPage.url();
      expect(url).toContain('/docs');

      await newPage.close();
```
