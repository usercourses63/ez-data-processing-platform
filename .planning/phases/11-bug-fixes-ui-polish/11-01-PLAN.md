```css
/* =============================================
   PHASE 11: RTL Bug Fixes for Ant Design 5.x
   Known issues: #40090 (Badge), #52942 (Table), #52371 (Collapse)
   ============================================= */

/* FIX: Table fixed columns shadow in RTL (GitHub #52942)
   Shadow appears on wrong side for fixed columns in RTL mode */
.rtl .ant-table-cell-fix-left::after {
  box-shadow: none !important;
}
.rtl .ant-table-cell-fix-right::after {
  box-shadow: inset -10px 0 8px -8px rgba(5, 5, 5, 0.06) !important;
}

/* FIX: Badge positioning in RTL (GitHub #40090)
   Badge count appears on wrong side in RTL mode */
.rtl .ant-badge .ant-badge-count,
.rtl .ant-badge .ant-badge-dot {
  right: auto !important;
  left: 0 !important;
  transform: translate(-50%, -50%) !important;
}

/* FIX: Collapse arrow direction in RTL (GitHub #52371)
   Arrow icons point wrong direction in RTL mode */
.rtl .ant-collapse-arrow {
  transform: rotate(180deg) !important;
}
.rtl .ant-collapse-item-active > .ant-collapse-header .ant-collapse-arrow {
  transform: rotate(270deg) !important;
}

/* FIX: Sider/Menu submenu arrow direction in RTL */
.rtl .ant-menu-submenu-expand-icon,
.rtl .ant-menu-submenu-arrow {
  right: auto !important;
  left: 16px !important;
}

/* FIX: Sider trigger icon direction in RTL */
.rtl .ant-layout-sider-trigger {
  text-align: center;
}
.rtl .ant-layout-sider-collapsed .anticon-left,
.rtl .ant-layout-sider-collapsed .anticon-right {
  transform: rotate(180deg);
}
```

Place this section after the existing RTL overrides section (around line 400+) and before the MOCKUP COMPONENT STYLING section.
