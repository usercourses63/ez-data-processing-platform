using DataProcessing.Shared.Connectors;
using DataProcessing.Shared.Entities;
using FluentAssertions;
using Microsoft.Extensions.Logging.Abstractions;
using Xunit;
using Xunit.Abstractions;

namespace DataProcessing.IntegrationTests.Connectors;

/// <summary>
/// Local file connector integration tests.
///
/// IMPORTANT: These tests are marked OCP-INCOMPATIBLE because:
/// 1. OCP pods run with readOnlyRootFilesystem: true
/// 2. Local paths like C:\ or /tmp don't exist in container context
/// 3. Local connector is for development/testing only, not production
///
/// These tests serve as:
/// - Reference implementation showing connector test patterns
/// - Local development validation
/// - Documentation of LocalFileConnector behavior
///
/// In OCP deployments, use NFS connector via NAS devices for file access.
/// </summary>
[Collection("Integration")]
[Trait("OCP", "Incompatible")]
public class LocalFileConnectorTests
{
    private readonly ITestOutputHelper _output;
    private readonly LocalFileConnector _connector;
    private readonly string _testDirectory;
    private readonly string _testRunId;

    public LocalFileConnectorTests(ITestOutputHelper output)
    {
        _output = output;
        _connector = new LocalFileConnector(NullLogger<LocalFileConnector>.Instance);
        _testRunId = Guid.NewGuid().ToString("N")[..8];
        _testDirectory = Path.Combine(Path.GetTempPath(), $"ez-local-test-{_testRunId}");
        Directory.CreateDirectory(_testDirectory);
    }

    // Cleanup test directory after each test
    public void Dispose()
    {
        if (Directory.Exists(_testDirectory))
        {
            Directory.Delete(_testDirectory, recursive: true);
        }
    }

    // Test methods...
}

# Protocol Test Coverage Guide

This guide documents how to run and extend the EZ Platform protocol connector tests.

## Overview

EZ Platform supports 8 file/message protocols:
| Protocol | Connector | Test Class | OCP Compatible |
|----------|-----------|------------|----------------|
| FTP | FtpConnector | FtpConnectorTests | Yes |
| SFTP | SftpConnector | SftpConnectorTests | Yes |
| HTTP/WebDAV | HttpApiConnector | HttpApiConnectorTests | Yes |
| S3/MinIO | S3Connector | S3ConnectorTests | Yes |
| SMB | SmbConnector | SmbConnectorTests | Yes (with tunnel) |
| NFS | NfsConnector | NfsConnectorTests | Yes (via PVC) |
| Kafka | KafkaConnector | KafkaConnectorTests | Yes |
| Local | LocalFileConnector | LocalFileConnectorTests | No |

## Prerequisites

### File-Simulator Setup
Most protocol tests run against file-simulator. See: [file-simulator-setup.md](./file-simulator-setup.md)

1. Ensure file-simulator cluster is running: `minikube status -p file-simulator`
2. Get file-simulator IP: `minikube ip -p file-simulator`
3. Set environment variable: `$env:FILE_SIMULATOR_IP = "172.25.x.x"`

### Kafka Setup
Kafka tests require port-forward to ez-platform Kafka:
```powershell
kubectl port-forward svc/kafka 9094:9094 -n ez-platform
```

## Running Protocol Tests

### Run All Protocol Tests
```powershell
dotnet test tests/IntegrationTests --filter "Category=Protocol"
```

### Run Specific Protocol
```powershell
# FTP only
dotnet test tests/IntegrationTests --filter "Protocol=FTP"

# Multiple protocols
dotnet test tests/IntegrationTests --filter "Protocol=FTP|Protocol=SFTP|Protocol=HTTP"

# All OCP-compatible
dotnet test tests/IntegrationTests --filter "Category=Protocol&OCP=Compatible"
```

### Run with Verbose Output
```powershell
dotnet test tests/IntegrationTests --filter "Category=Protocol" --logger "console;verbosity=detailed"
```

## Protocol-Specific Notes

### FTP, SFTP, HTTP (Standard)
- NodePort services on file-simulator
- No special setup beyond FILE_SIMULATOR_IP

### S3/MinIO
- Uses MinIO S3-compatible endpoint
- Port 30900 on file-simulator
- Default bucket: `ez-data`
- Credentials: minioadmin/minioadmin123

### SMB (Requires Tunnel)
- SMB protocol requires port 445 (not remappable)
- Must run `minikube tunnel -p file-simulator`
- Tests skip with message if tunnel not active
- Share name: `files`
- Credentials: smbuser/smbpass123

### NFS (Requires K8s Pod Context)
- NFS connector accesses mounted PVC paths
- Mounts only exist inside K8s pods
- Set `NFS_MOUNT_PATH=/mnt/nas-input-1` to enable tests
- For full testing, run tests inside K8s pod:
  ```yaml
  apiVersion: v1
  kind: Pod
  metadata:
    name: protocol-test-pod
  spec:
    containers:
    - name: tests
      image: mcr.microsoft.com/dotnet/sdk:10.0
      env:
      - name: NFS_MOUNT_PATH
        value: /mnt/nas-input-1
      volumeMounts:
      - name: nas-input
        mountPath: /mnt/nas-input-1
    volumes:
    - name: nas-input
      persistentVolumeClaim:
        claimName: nas-input-1-pvc
  ```

### Kafka
- Requires port-forward to Kafka broker
- Tests use unique consumer groups to avoid conflicts
- Topic names include test run ID

### Local (OCP-Incompatible)
- Reference tests only
- Uses local temp directory
- Will not work in OCP (readOnlyRootFilesystem)
- Use NFS connector for production file access

## Test Traits

| Trait | Values | Purpose |
|-------|--------|---------|
| Category | Protocol | All protocol connector tests |
| Protocol | FTP, SFTP, HTTP, S3, SMB, NFS, Kafka, Local | Specific protocol |
| OCP | Compatible, Incompatible | OCP deployment compatibility |
| Dependency | FileSimulator, Tunnel, K8sPod | External requirements |

## Adding New Protocol Tests

1. Create test class in `tests/IntegrationTests/Connectors/`
2. Add `[Collection("Integration")]` for shared fixtures
3. Add `[Trait("Category", "Protocol")]` and `[Trait("Protocol", "XXX")]`
4. Implement skip logic for unavailable dependencies
5. Follow existing patterns for TestConnection, WriteRead, ListFiles
6. Update this guide with new protocol details

## Troubleshooting

### Tests Skip with "File-simulator not available"
1. Check file-simulator status: `minikube status -p file-simulator`
2. Verify IP: `minikube ip -p file-simulator`
3. Set environment: `$env:FILE_SIMULATOR_IP = "<ip>"`
4. Test connectivity: `Test-NetConnection -ComputerName $env:FILE_SIMULATOR_IP -Port 30021`

### SMB Tests Skip with "requires minikube tunnel"
1. Run tunnel in separate terminal: `minikube tunnel -p file-simulator`
2. Verify port 445 is forwarded
3. Keep tunnel running during test execution

### NFS Tests Skip with "NFS mount not available"
1. Tests require K8s pod context with mounted PVC
2. For local dev, run: `$env:NFS_MOUNT_PATH = "C:\simulator-data\nas-input-1"`
3. For full testing, deploy test pod with PVC mounts

### Kafka Tests Fail with "broker not available"
1. Verify port-forward: `kubectl port-forward svc/kafka 9094:9094 -n ez-platform`
2. Check broker health: `kubectl logs deploy/kafka -n ez-platform`
