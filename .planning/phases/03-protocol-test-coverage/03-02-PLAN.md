```csharp
using DataProcessing.Shared.Entities;
using DataProcessing.Shared.Services;
using Microsoft.Extensions.Logging;
using SMBLibrary;
using SMBLibrary.Client;
using System.Diagnostics;
using System.Net;

namespace DataProcessing.Shared.Connectors;

/// <summary>
/// Connector for reading/writing files via SMB protocol.
/// Uses SMBLibrary for SMB 2/3 support.
/// Note: SMB requires port 445; NodePort mapping doesn't work.
/// Use minikube tunnel for testing against file-simulator.
/// </summary>
public class SmbConnector : IDataSourceConnector, IServerConnector
{
    private readonly ILogger<SmbConnector> _logger;
    public string ConnectorType => "smb";

    // Constructor
    // IDataSourceConnector: ReadFileAsync, ListFilesAsync, TestConnectionAsync, GetFileMetadataAsync
    // IServerConnector: TestConnectionAsync, ListFilesAsync, ReadFileAsync, WriteFileAsync
    // Use SMB2Client for connections
    // Login with domain (empty string), username, password
    // Access shares via ISMBFileStore
}
```

Key implementation details:
- Use SMB2Client (not SMB1Client) for modern compatibility
- SMBTransportType.DirectTCPTransport for port 445
- Login returns NTStatus - check for STATUS_SUCCESS
- Get file store via client.TreeConnect(shareName, out ISMBFileStore)
- File operations via store.CreateFile, store.ReadFile, store.WriteFile
- Always close file handles, tree, and logoff/disconnect
- Handle UnauthorizedAccessException for credential errors
- Handle SocketException for connectivity errors
