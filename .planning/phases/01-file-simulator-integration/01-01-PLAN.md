<task type="auto">
  <name>Task 3: Create connectivity verification script</name>
  <files>scripts/verify-file-simulator.ps1</files>
  <action>
Create a PowerShell script that:
1. Gets the file-simulator Minikube IP
2. Updates the ConfigMap with the actual IP (replaces placeholder)
3. Applies the ConfigMap to ez-platform namespace
4. Tests connectivity to each protocol endpoint
5. Reports status summary

```powershell
# verify-file-simulator.ps1
# Verifies cross-cluster connectivity from EZ Platform to file-simulator

param(
    [switch]$Apply,      # Apply ConfigMap after updating IP
    [switch]$TestOnly    # Only test connectivity, don't update ConfigMap
)

$ErrorActionPreference = "Stop"

# Get file-simulator Minikube IP
Write-Host "`n=== Getting file-simulator IP ===" -ForegroundColor Cyan
$simulatorIp = minikube ip -p file-simulator 2>&1
if ($LASTEXITCODE -ne 0) {
    Write-Host "ERROR: file-simulator cluster not running. Start with:" -ForegroundColor Red
    Write-Host "  minikube start --profile file-simulator --driver=hyperv --memory=8192 --cpus=4" -ForegroundColor Yellow
    exit 1
}
Write-Host "File-simulator IP: $simulatorIp" -ForegroundColor Green

if (-not $TestOnly) {
    # Update ConfigMap with actual IP
    Write-Host "`n=== Updating ConfigMap ===" -ForegroundColor Cyan
    $configMapPath = Join-Path $PSScriptRoot "..\k8s\configmaps\file-simulator-config.yaml"
    $content = Get-Content $configMapPath -Raw
    $updated = $content -replace '\{\{FILE_SIMULATOR_IP\}\}', $simulatorIp

    # Write to temp file for applying
    $tempPath = Join-Path $env:TEMP "file-simulator-config-resolved.yaml"
    $updated | Out-File -FilePath $tempPath -Encoding utf8
    Write-Host "ConfigMap updated with IP: $simulatorIp"

    if ($Apply) {
        Write-Host "`n=== Applying ConfigMap ===" -ForegroundColor Cyan
        kubectl --context=minikube apply -f $tempPath -n ez-platform
        if ($LASTEXITCODE -eq 0) {
            Write-Host "ConfigMap applied successfully" -ForegroundColor Green
        } else {
            Write-Host "Failed to apply ConfigMap" -ForegroundColor Red
            exit 1
        }
    }
}

# Test connectivity
Write-Host "`n=== Testing Connectivity ===" -ForegroundColor Cyan

$tests = @(
    @{ Name = "FTP";     Port = 30021; Description = "FTP Server" },
    @{ Name = "SFTP";    Port = 30022; Description = "SFTP Server" },
    @{ Name = "S3/MinIO"; Port = 30900; Description = "S3-compatible API" },
    @{ Name = "HTTP";    Port = 30088; Description = "HTTP Server" },
    @{ Name = "WebDAV";  Port = 30089; Description = "WebDAV Server" }
)

$results = @()
foreach ($test in $tests) {
    $result = Test-NetConnection -ComputerName $simulatorIp -Port $test.Port -WarningAction SilentlyContinue
    $status = if ($result.TcpTestSucceeded) { "OK" } else { "FAIL" }
    $color = if ($result.TcpTestSucceeded) { "Green" } else { "Red" }

    Write-Host "  $($test.Name.PadRight(10)) : $status - $($test.Description)" -ForegroundColor $color
    $results += @{ Test = $test.Name; Success = $result.TcpTestSucceeded }
}

# Test S3 health endpoint
Write-Host "`n=== Testing S3 Health ===" -ForegroundColor Cyan
try {
    $s3Response = Invoke-WebRequest -Uri "http://${simulatorIp}:30900/minio/health/live" -Method GET -TimeoutSec 5 -UseBasicParsing
    if ($s3Response.StatusCode -eq 200) {
        Write-Host "  S3 Health  : OK - MinIO responding" -ForegroundColor Green
        $results += @{ Test = "S3 Health"; Success = $true }
    }
} catch {
    Write-Host "  S3 Health  : FAIL - $_" -ForegroundColor Red
    $results += @{ Test = "S3 Health"; Success = $false }
}

# Summary
Write-Host "`n=== Summary ===" -ForegroundColor Cyan
$passed = ($results | Where-Object { $_.Success }).Count
$total = $results.Count
$color = if ($passed -eq $total) { "Green" } else { "Yellow" }
Write-Host "Results: $passed/$total tests passed" -ForegroundColor $color

if ($passed -lt $total) {
    Write-Host "`nTroubleshooting:" -ForegroundColor Yellow
    Write-Host "1. Ensure file-simulator is deployed: helm list --kube-context=file-simulator -n file-simulator"
    Write-Host "2. Check pods are running: kubectl --context=file-simulator get pods -n file-simulator"
    Write-Host "3. Verify NodePort services: kubectl --context=file-simulator get svc -n file-simulator"
    exit 1
}

Write-Host "`nFile-simulator connectivity verified successfully!" -ForegroundColor Green
```
