    1. **global-setup.ts** - Pre-flight connectivity checks:
       ```typescript
       import { request } from '@playwright/test';

       async function globalSetup() {
         const SIMULATOR_BASE = process.env.FILE_SIMULATOR_URL || 'http://172.17.89.141:30500';
         const EZ_API_BASE = process.env.EZ_API_URL || 'http://localhost:5001';

         const ctx = await request.newContext();

         // Check file-simulator health
         console.log(`[Global Setup] Checking file-simulator at ${SIMULATOR_BASE}...`);
         const simHealth = await ctx.get(`${SIMULATOR_BASE}/health`).catch(() => null);
         if (!simHealth || !simHealth.ok()) {
           throw new Error(`FATAL: File-simulator unavailable at ${SIMULATOR_BASE}. Protocol tests require file-simulator.`);
         }
         console.log('[Global Setup] File-simulator: HEALTHY');

         // Check EZ API health
         console.log(`[Global Setup] Checking EZ API at ${EZ_API_BASE}...`);
         const ezHealth = await ctx.get(`${EZ_API_BASE}/health`).catch(() => null);
         if (!ezHealth || !ezHealth.ok()) {
           throw new Error(`FATAL: EZ API unavailable at ${EZ_API_BASE}. Protocol tests require EZ backend.`);
         }
         console.log('[Global Setup] EZ API: HEALTHY');

         // Check file-operations endpoint exists (from Plan 01)
         console.log('[Global Setup] Verifying file-operations endpoint...');
         const fileOpsCheck = await ctx.post(`${EZ_API_BASE}/api/v1/file-operations/nonexistent/list`, {
           data: { Pattern: '*' }
         }).catch(() => null);
         // Expect 404 (server not found) not 405 (route not found)
         if (fileOpsCheck && fileOpsCheck.status() === 405) {
           throw new Error('FATAL: File-operations endpoint not deployed. Rebuild DataSourceManagementService.');
         }
         console.log('[Global Setup] File-operations endpoint: AVAILABLE');

         // Discover simulator servers for test summary
         const serversResp = await ctx.get(`${SIMULATOR_BASE}/api/servers`);
         const servers = await serversResp.json();
         const protocols = [...new Set(servers.map((s: any) => s.Protocol))];
         console.log(`[Global Setup] Simulator protocols available: ${protocols.join(', ')}`);
         console.log(`[Global Setup] Total simulator servers: ${servers.length}`);

         await ctx.dispose();
       }

       export default globalSetup;
       ```

    2. **global-teardown.ts** - Generate final matrix report:
       ```typescript
       import { matrixReporter } from '../helpers/report-generator';
       import * as path from 'path';

       async function globalTeardown() {
         const reportPath = path.join(__dirname, '..', '..', '..', 'test-results', 'protocol-matrix-report.md');
         console.log(`[Global Teardown] Writing protocol matrix report to ${reportPath}`);
         matrixReporter.writeReport(reportPath);
         console.log('[Global Teardown] Report generated successfully');

         // Print summary to console
         const markdown = matrixReporter.generateMarkdown();
         console.log('\n' + '='.repeat(80));
         console.log('PROTOCOL x FORMAT MATRIX REPORT');
         console.log('='.repeat(80));
         console.log(markdown);
       }

       export default globalTeardown;
       ```

       NOTE: The matrixReporter singleton must persist across test files. If Playwright runs tests in separate workers, the singleton won't share state. SOLUTION: Use a file-based accumulation strategy:
       - Each test file writes results to a temp JSON file (e.g., `test-results/.protocol-results-{protocol}.json`)
       - global-teardown reads all temp JSON files and combines into the final markdown report
       - Update `report-generator.ts` if needed to support file-based persistence (add `saveResults(path)` and `loadResults(path)` methods)
       - Alternative: Use Playwright's `process.env` or file-system to share state if running with `workers: 1`
