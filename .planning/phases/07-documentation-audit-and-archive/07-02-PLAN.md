```powershell
# Session file identification script
# Identifies session files by pattern AND extracts their month from LastWriteTime

$sessionPatterns = @(
    '*SESSION*',
    '*DAY*-COMPLETE*',
    '*-DAY-*'
)

$sessionDirectories = @(
    'docs/planning/Phase-MVP-Deployment',
    'docs/planning/InvalidRecords',
    'docs/planning/Task-3-Metrics'
)

# Find session files by name pattern
$sessionsByPattern = Get-ChildItem -Path "." -Filter "*.md" -Recurse |
    Where-Object { $_.FullName -notmatch 'node_modules|\.git|\.planning' } |
    Where-Object {
        $name = $_.Name
        $sessionPatterns | Where-Object { $name -like $_ }
    }

# Find session files by directory
$sessionsByDir = Get-ChildItem -Path $sessionDirectories -Filter "*.md" -ErrorAction SilentlyContinue

# Combine and deduplicate
$allSessions = @($sessionsByPattern) + @($sessionsByDir) |
    Sort-Object FullName -Unique

# Group by month based on LastWriteTime
$sessionsByMonth = $allSessions | Group-Object { $_.LastWriteTime.ToString('yyyy-MM') }

# Generate git mv commands
foreach ($monthGroup in $sessionsByMonth) {
    $month = $monthGroup.Name
    Write-Host "### $month"
    foreach ($file in $monthGroup.Group) {
        $relativePath = $file.FullName -replace [regex]::Escape((Get-Location).Path + '\'), ''
        Write-Host "git mv `"$relativePath`" `"docs/archive/sessions/$month/`""
    }
}
```

**Step 5: Generate session logs index:**
Create `docs/archive/sessions/index.md` by parsing each session file:

```powershell
# Index generation script
$sessions = Get-ChildItem -Path "docs/archive/sessions" -Filter "*.md" -Recurse |
    Where-Object { $_.Name -ne "index.md" } |
    Sort-Object { $_.Directory.Name }, Name

$indexContent = @"
# Archived Session Logs

This directory contains archived session logs from development work.
These files are preserved for historical reference but are no longer active documentation.

## Archive Date
Archived: 2026-02-02

## Session Logs by Month

"@

# Group by month folder
$byMonth = $sessions | Group-Object { $_.Directory.Name }

foreach ($monthGroup in ($byMonth | Sort-Object Name)) {
    $month = $monthGroup.Name
    $indexContent += "`n### $month`n"
    $indexContent += "| Session | Description | Original Location |`n"
    $indexContent += "|---------|-------------|-------------------|`n"

    foreach ($session in $monthGroup.Group) {
        # Extract first heading from file as description
        $content = Get-Content $session.FullName -First 10 | Out-String
        $heading = if ($content -match '^#\s+(.+)$') { $Matches[1] } else { "Session log" }
        $heading = $heading.Substring(0, [Math]::Min(50, $heading.Length))

        $link = "[$($session.BaseName)](./$month/$($session.Name))"
        $indexContent += "| $link | $heading | [from inventory] |`n"
    }
}

$indexContent += @"

## Total Archived
$(($byMonth | ForEach-Object { "- $($_.Name): $($_.Count) files" }) -join "`n")
- **Total: $($sessions.Count) files**

---
*Index generated: $(Get-Date -Format 'yyyy-MM-dd')*
"@

$indexContent | Out-File -FilePath "docs/archive/sessions/index.md" -Encoding UTF8
```

```powershell
# Broken link detection script
$docsDir = "docs"
$mdFiles = Get-ChildItem -Path $docsDir -Filter "*.md" -Recurse
$brokenLinks = @()

foreach ($file in $mdFiles) {
    $content = Get-Content $file.FullName -Raw

    # Match markdown links: [text](path.md) or [text](./path.md)
    $links = [regex]::Matches($content, '\[([^\]]+)\]\(([^)]+\.md)\)')

    foreach ($link in $links) {
        $linkPath = $link.Groups[2].Value

        # Skip external URLs
        if ($linkPath -match '^https?://') { continue }

        # Resolve relative path
        $fullPath = if ($linkPath.StartsWith('./') -or $linkPath.StartsWith('../')) {
            Join-Path $file.DirectoryName $linkPath
        } else {
            Join-Path $docsDir $linkPath
        }

        if (-not (Test-Path $fullPath)) {
            $brokenLinks += [PSCustomObject]@{
                SourceFile = $file.FullName -replace [regex]::Escape("$docsDir\"), ''
                BrokenLink = $linkPath
                LinkedText = $link.Groups[1].Value
            }
        }
    }
}

if ($brokenLinks.Count -gt 0) {
    Write-Host "Found $($brokenLinks.Count) broken links:"
    $brokenLinks | Format-Table -AutoSize
} else {
    Write-Host "No broken links found"
}
```
