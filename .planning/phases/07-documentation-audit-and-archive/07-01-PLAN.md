Create and run a PowerShell script that reads docs-inventory.csv, applies categorization rules, and writes the updated CSV:

```powershell
# File: scripts/categorize-docs.ps1
# Reads docs-inventory.csv, applies rules, updates Category column

$inventory = Import-Csv -Path "docs-inventory.csv"

foreach ($file in $inventory) {
    $path = $file.Path
    $name = Split-Path $path -Leaf
    $dir = $file.Directory

    # Default to KEEP
    $category = "KEEP"

    # Rule 1: Session logs -> ARCHIVE
    if ($name -match 'SESSION|DAY.*COMPLETE|DAY-\d+') {
        $category = "ARCHIVE"
    }
    # Rule 2: docs/planning/ entire directory -> ARCHIVE
    elseif ($dir -match '^docs[/\\]planning') {
        $category = "ARCHIVE"
    }
    # Rule 3: Root markdown files (check by directory being project root or empty)
    elseif ($dir -eq '' -or $dir -eq '.') {
        if ($name -match '^(README|CLAUDE|CHANGELOG)\.md$') {
            $category = "KEEP"
        }
        elseif ($name -match 'RELEASE|COMPLETION-REPORT|FINAL-DEPLOYMENT|VERIFICATION-REPORT') {
            $category = "ARCHIVE"  # Release artifacts
        }
        elseif ($name -match 'DEPLOYMENT-PLAN|DEPLOYMENT-.*STATUS') {
            $category = "ARCHIVE"  # Deployment guides -> will move to docs/deployment/
        }
        elseif ($name -match 'Presentation|SYNC|RECOVERY|GIT-|PRODUCTION-SYNC') {
            $category = "ARCHIVE"  # Presentations, operational logs
        }
        elseif ($name -match 'STATUS|REPORT|SUMMARY') {
            $category = "ARCHIVE"  # Status reports
        }
        else {
            $category = "ARCHIVE"  # Other root files -> archive
        }
    }
    # Rule 4: Service/component READMEs -> KEEP
    elseif ($dir -match '^src[/\\]Services|^src[/\\]Frontend|^k8s|^tools|^test-data') {
        $category = "KEEP"
    }
    # Rule 5: docs/ reference documentation -> KEEP
    elseif ($dir -match '^docs[/\\](reference|architecture|testing|admin|deployment|design-system|frontend|installation|maintenance|user-guide)') {
        $category = "KEEP"
    }
    elseif ($path -eq 'docs/DOCUMENTATION-INDEX.md') {
        $category = "KEEP"
    }
    # Rule 6: Already in archive -> KEEP (in archive)
    elseif ($dir -match 'archive') {
        $category = "KEEP"
    }
    # Rule 7: DELETE candidates - duplicates or deprecated
    elseif ($name -match 'old|deprecated|backup' -and $dir -match 'archive') {
        $category = "DELETE"
    }

    $file.Category = $category
}

# Export updated CSV
$inventory | Export-Csv -Path "docs-inventory.csv" -NoTypeInformation
Write-Host "Categorization complete. Summary:"
$inventory | Group-Object Category | Format-Table Name, Count
```

Generate using this PowerShell script to extract exact paths and months:
```powershell
# Script to generate session-logs-archive-plan.md with explicit paths
$inventory = Import-Csv -Path "docs-inventory.csv"

# Filter session files (files in docs/planning/ subdirs + session patterns)
$sessionFiles = $inventory | Where-Object {
    $_.Path -match 'SESSION|DAY.*COMPLETE|DAY-\d+' -or
    $_.Directory -match 'docs[/\\]planning[/\\](Phase-MVP|InvalidRecords|Task-3)'
}

# Group by month (extract from LastModified column)
$byMonth = $sessionFiles | Group-Object {
    $date = [DateTime]::Parse($_.LastModified)
    $date.ToString('yyyy-MM')
}

# Generate markdown content
$content = @"
# Session Logs Archive Plan

## Summary
- Total session logs: $($sessionFiles.Count)
- Target directory: docs/archive/sessions/
- Organized by: LastModified month

## EXPLICIT File List with Git Commands

**IMPORTANT: Plan 02 Task 1 will execute these commands directly - no re-identification needed.**

"@

foreach ($monthGroup in ($byMonth | Sort-Object Name)) {
    $month = $monthGroup.Name
    $content += "`n### $month`n"
    $content += "| Current Path | Git Command |`n"
    $content += "|--------------|-------------|`n"

    foreach ($file in $monthGroup.Group) {
        $srcPath = $file.Path -replace '\\', '/'
        $fileName = Split-Path $srcPath -Leaf
        $gitCmd = "git mv `"$srcPath`" `"docs/archive/sessions/$month/$fileName`""
        $content += "| $srcPath | ``$gitCmd`` |`n"
    }

    $content += "`n**Batch command for $month:**`n"
    $content += "```bash`n"
    $content += "mkdir -p docs/archive/sessions/$month`n"
    foreach ($file in $monthGroup.Group) {
        $srcPath = $file.Path -replace '\\', '/'
        $fileName = Split-Path $srcPath -Leaf
        $content += "git mv `"$srcPath`" `"docs/archive/sessions/$month/`"`n"
    }
    $content += "git commit -m `"docs(07-02): archive session logs from $month`"`n"
    $content += "```n"
}

$content | Out-File -FilePath ".planning/phases/07-documentation-audit-and-archive/session-logs-archive-plan.md" -Encoding UTF8
```
