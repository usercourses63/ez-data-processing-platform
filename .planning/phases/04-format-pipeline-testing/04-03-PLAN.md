// FileVolumeTests.cs - Load Testing for File Volume Throughput
// Tests 100-1000 file processing scenarios
// Establishes performance baselines for regression tracking

using Xunit;
using Xunit.SkippableFact;
using FluentAssertions;
using System.Diagnostics;

namespace DataProcessing.IntegrationTests.LoadTests;

[Collection("Integration")]
public class FileVolumeTests
{
    private readonly PipelineFixture _pipeline;
    private readonly KafkaFixture _kafka;
    private readonly FileSimulatorFixture _fileSimulator;

    public FileVolumeTests(
        PipelineFixture pipeline,
        KafkaFixture kafka,
        FileSimulatorFixture fileSimulator)
    {
        _pipeline = pipeline;
        _kafka = kafka;
        _fileSimulator = fileSimulator;
    }

    private void SkipIfUnavailable()
    {
        Skip.If(!_pipeline.IsAvailable, _pipeline.SkipReason);
        Skip.If(!_fileSimulator.IsAvailable, "File-simulator required for load tests");
    }

    [SkippableFact]
    [Trait("Category", "LoadTest")]
    [Trait("FileCount", "100")]
    public async Task LoadTest_100Files_ProcessesWithinTimeout()
    {
        SkipIfUnavailable();

        // Arrange
        const int fileCount = 100;
        const int recordsPerFile = 10; // Keep small for volume testing
        var correlationId = Guid.NewGuid().ToString();
        var stagedFiles = new List<string>();
        var stopwatch = Stopwatch.StartNew();

        Console.WriteLine($"Starting 100-file load test: {correlationId}");

        // Stage files via FTP
        for (int i = 0; i < fileCount; i++)
        {
            var content = TestDataFactory.GenerateCsv(recordsPerFile);
            var fileName = $"load-{correlationId[..8]}-{i:D3}.csv";
            await _pipeline.StageTestFileAsync(fileName, content);
            stagedFiles.Add(fileName);

            if (i % 20 == 0)
                Console.WriteLine($"  Staged {i + 1}/{fileCount} files...");
        }

        var stagingDuration = stopwatch.Elapsed;
        Console.WriteLine($"Staging complete: {stagingDuration.TotalSeconds:F1}s");

        // Act - Trigger file discovery for all files
        stopwatch.Restart();
        var processingStartTime = DateTime.UtcNow;

        foreach (var fileName in stagedFiles)
        {
            var fileEvent = CreateFileDiscoveredEvent(correlationId, fileName);
            await _kafka.ProduceAsync(
                TestConfiguration.KafkaTopics.FileDiscovered,
                $"{correlationId}-{fileName}",
                fileEvent);
        }

        Console.WriteLine($"Published {fileCount} FileDiscoveredEvents");

        // Assert - Wait for all files to be processed
        var processedCount = await WaitForProcessedCountAsync(
            fileCount,
            TimeSpan.FromMinutes(5));

        var processingDuration = stopwatch.Elapsed;

        // Log results
        Console.WriteLine($"Load Test Results (100 files):");
        Console.WriteLine($"  Files processed: {processedCount}/{fileCount}");
        Console.WriteLine($"  Processing time: {processingDuration.TotalSeconds:F1}s");
        Console.WriteLine($"  Throughput: {fileCount / processingDuration.TotalMinutes:F1} files/min");

        // Assertions
        processedCount.Should().Be(fileCount, "all files should be processed");
        processingDuration.Should().BeLessThan(
            TimeSpan.FromMinutes(5),
            "100 files should complete within 5 minutes");

        // Cleanup
        foreach (var fileName in stagedFiles)
        {
            await CleanupTestFileAsync(fileName);
        }
    }

    [SkippableFact]
    [Trait("Category", "LoadTest")]
    [Trait("FileCount", "1000")]
    public async Task LoadTest_1000Files_ProcessesWithinTimeout()
    {
        SkipIfUnavailable();

        // Stretch goal - 1000 files
        const int fileCount = 1000;
        const int recordsPerFile = 5; // Smaller for volume
        var correlationId = Guid.NewGuid().ToString();
        var stagedFiles = new List<string>();
        var stopwatch = Stopwatch.StartNew();

        Console.WriteLine($"Starting 1000-file load test (stretch goal): {correlationId}");

        // Stage files in batches to avoid overwhelming FTP
        const int batchSize = 100;
        for (int batch = 0; batch < fileCount / batchSize; batch++)
        {
            var batchTasks = new List<Task>();
            for (int i = 0; i < batchSize; i++)
            {
                var fileIndex = batch * batchSize + i;
                var content = TestDataFactory.GenerateCsv(recordsPerFile);
                var fileName = $"load1k-{correlationId[..8]}-{fileIndex:D4}.csv";
                batchTasks.Add(_pipeline.StageTestFileAsync(fileName, content)
                    .ContinueWith(_ => stagedFiles.Add(fileName)));
            }
            await Task.WhenAll(batchTasks);
            Console.WriteLine($"  Staged batch {batch + 1}/{fileCount / batchSize}");
        }

        var stagingDuration = stopwatch.Elapsed;
        Console.WriteLine($"Staging complete: {stagingDuration.TotalMinutes:F1} min");

        // Act - Publish all events
        stopwatch.Restart();
        var publishTasks = stagedFiles.Select(fileName =>
            _kafka.ProduceAsync(
                TestConfiguration.KafkaTopics.FileDiscovered,
                $"{correlationId}-{fileName}",
                CreateFileDiscoveredEvent(correlationId, fileName)));
        await Task.WhenAll(publishTasks);

        Console.WriteLine($"Published {fileCount} FileDiscoveredEvents");

        // Assert - Wait for processing (30 minute timeout for 1000 files)
        var processedCount = await WaitForProcessedCountAsync(
            fileCount,
            TimeSpan.FromMinutes(30));

        var processingDuration = stopwatch.Elapsed;

        Console.WriteLine($"Load Test Results (1000 files):");
        Console.WriteLine($"  Files processed: {processedCount}/{fileCount}");
        Console.WriteLine($"  Processing time: {processingDuration.TotalMinutes:F1} min");
        Console.WriteLine($"  Throughput: {fileCount / processingDuration.TotalMinutes:F1} files/min");

        // Stretch goal - may not complete fully
        processedCount.Should().BeGreaterOrEqualTo(
            (int)(fileCount * 0.95),
            "at least 95% of files should be processed");
    }

    [SkippableFact]
    [Trait("Category", "LoadTest")]
    [Trait("FileCount", "10")]
    public async Task LoadTest_QuickBaseline_EstablishesMetrics()
    {
        SkipIfUnavailable();

        // Quick test for baseline metrics
        const int fileCount = 10;
        const int recordsPerFile = 50;
        var correlationId = Guid.NewGuid().ToString();
        var stopwatch = Stopwatch.StartNew();

        // Stage and process
        for (int i = 0; i < fileCount; i++)
        {
            var content = TestDataFactory.GenerateCsv(recordsPerFile);
            var fileName = $"baseline-{correlationId[..8]}-{i:D2}.csv";
            await _pipeline.StageTestFileAsync(fileName, content);
            await _kafka.ProduceAsync(
                TestConfiguration.KafkaTopics.FileDiscovered,
                $"{correlationId}-{fileName}",
                CreateFileDiscoveredEvent(correlationId, fileName));
        }

        var processedCount = await WaitForProcessedCountAsync(
            fileCount,
            TimeSpan.FromMinutes(2));

        var duration = stopwatch.Elapsed;

        Console.WriteLine($"Baseline Results:");
        Console.WriteLine($"  Files: {processedCount}/{fileCount}");
        Console.WriteLine($"  Duration: {duration.TotalSeconds:F1}s");
        Console.WriteLine($"  Per-file: {duration.TotalMilliseconds / fileCount:F0}ms");

        processedCount.Should().Be(fileCount);
    }

    private string CreateFileDiscoveredEvent(string correlationId, string fileName)
    {
        return System.Text.Json.JsonSerializer.Serialize(new
        {
            CorrelationId = correlationId,
            DataSourceId = _pipeline.TestDataSourceId,
            FileName = fileName,
            FilePath = $"/input/{fileName}",
            Timestamp = DateTime.UtcNow,
            PublishedBy = "LoadTest",
            MessageVersion = 1
        });
    }

    private async Task<int> WaitForProcessedCountAsync(int expected, TimeSpan timeout)
    {
        var deadline = DateTime.UtcNow.Add(timeout);
        var processedCount = 0;

        using var consumer = _kafka.CreateConsumer(TestConfiguration.KafkaTopics.RecordOutput);

        while (DateTime.UtcNow < deadline && processedCount < expected)
        {
            var result = consumer.Consume(TimeSpan.FromSeconds(1));
            if (result != null)
            {
                processedCount++;
                if (processedCount % 50 == 0)
                    Console.WriteLine($"  Processed: {processedCount}/{expected}");
            }
        }

        return processedCount;
    }

    private async Task CleanupTestFileAsync(string fileName)
    {
        // Remove staged file from file-simulator
        // Implementation via FTP delete
    }
}

This guide documents the Phase 4 testing strategy for format conversions and pipeline integration in the EZ Data Processing Platform.

## Troubleshooting

### Common Issues

**"Kafka not available"**
- Verify port-forward: `kubectl port-forward svc/kafka 9094:9094 -n ez-platform`
- Check Kafka pods: `kubectl get pods -l app=kafka -n ez-platform`

**"File-simulator not available"**
- Set environment variable: `$env:FILE_SIMULATOR_IP = "your-ip"`
- Verify connectivity: `Test-NetConnection $env:FILE_SIMULATOR_IP -Port 30021`

**"EPPlus license exception"**
- Ensure test calls: `ExcelPackage.License.SetNonCommercialOrganization("EZ Platform Tests")`
- Check EPPlus version >= 8.0

**"Hebrew characters corrupted"**
- Verify UTF-8 encoding in stream operations
- Check `Encoding.UTF8.GetBytes()` usage
