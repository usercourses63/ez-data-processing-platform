```typescript
import { APIRequestContext } from '@playwright/test';
import { listFiles, readFile, DiscoveredFile } from '../../helpers/ez-api-client';

export async function waitForPipelineOutput(
  request: APIRequestContext,
  outputServerId: string,
  expectedFileCount: number,
  timeoutMs: number = 60000,
  pollIntervalMs: number = 3000
): Promise<DiscoveredFile[]> {
  const startTime = Date.now();
  let lastFiles: DiscoveredFile[] = [];
  while (Date.now() - startTime < timeoutMs) {
    try {
      lastFiles = await listFiles(request, outputServerId);
      if (lastFiles.length >= expectedFileCount) {
        return lastFiles;
      }
    } catch (err) {
      console.log(`[pipeline-poller] List files error (retrying): ${err}`);
    }
    await new Promise(r => setTimeout(r, pollIntervalMs));
  }
  throw new Error(
    `Pipeline output timeout after ${timeoutMs}ms: expected ${expectedFileCount} files, got ${lastFiles.length}`
  );
}

export function parseCsvContent(content: string): Record<string, string>[] {
  const lines = content.trim().split('\n');
  if (lines.length < 2) return [];
  const headers = lines[0].split(',').map(h => h.trim().replace(/^"|"$/g, ''));
  return lines.slice(1).map(line => {
    const values = line.split(',').map(v => v.trim().replace(/^"|"$/g, ''));
    const record: Record<string, string> = {};
    headers.forEach((h, i) => { record[h] = values[i] || ''; });
    return record;
  });
}

export async function compareOutputWithInput(
  request: APIRequestContext,
  outputServerId: string,
  outputFilePath: string,
  expectedRecordCount: number,
  sampleFields: string[],
  expectedSamples: Record<string, string>[]
): Promise<{ passed: boolean; details: string }> {
  const content = await readFile(request, outputServerId, outputFilePath);
  const records = parseCsvContent(content.toString('utf-8'));

  if (records.length !== expectedRecordCount) {
    return { passed: false, details: `Record count mismatch: expected ${expectedRecordCount}, got ${records.length}` };
  }

  for (let i = 0; i < Math.min(expectedSamples.length, records.length); i++) {
    for (const field of sampleFields) {
      if (records[i]?.[field] !== expectedSamples[i]?.[field]) {
        return { passed: false, details: `Field mismatch at record ${i}, field ${field}: expected '${expectedSamples[i]?.[field]}', got '${records[i]?.[field]}'` };
      }
    }
  }

  return { passed: true, details: `${records.length} records verified, ${sampleFields.length} fields sampled across ${expectedSamples.length} records` };
}
```

3. **validation-reporter.ts** — Adapt MatrixReporter pattern for E2E validation:
```typescript
import * as fs from 'fs';
import * as path from 'path';

const RESULTS_DIR = path.resolve(__dirname, '..', '..', '..', '..', 'test-results', '.e2e-validation-results');

export interface E2ETestResult {
  suite: string;       // 'pipeline-flow' | 'negative' | 'ui-workflow' | 'ui-monitoring' | 'cross-feature' | 'existing-backend' | 'existing-playwright'
  testName: string;
  passed: boolean;
  error?: string;
  durationMs?: number;
  category?: string;   // 'protocol' | 'format' | 'ui' | 'backend' | 'pipeline'
}

export class E2EValidationReporter {
  private results: E2ETestResult[] = [];
  private persistenceTag: string | null = null;

  enablePersistence(tag: string): void {
    this.persistenceTag = tag;
    if (!fs.existsSync(RESULTS_DIR)) {
      fs.mkdirSync(RESULTS_DIR, { recursive: true });
    }
  }

  addResult(result: E2ETestResult): void {
    this.results.push(result);
    if (this.persistenceTag) {
      this.appendResultToDisk(result);
    }
  }

  loadAllPersistedResults(): void {
    if (!fs.existsSync(RESULTS_DIR)) return;
    const files = fs.readdirSync(RESULTS_DIR).filter(f => f.endsWith('.json'));
    for (const file of files) {
      const content = fs.readFileSync(path.join(RESULTS_DIR, file), 'utf-8');
      try {
        const loaded: E2ETestResult[] = JSON.parse(content);
        this.results.push(...loaded);
      } catch {}
    }
  }

  static cleanPersistedResults(): void {
    if (!fs.existsSync(RESULTS_DIR)) return;
    const files = fs.readdirSync(RESULTS_DIR).filter(f => f.endsWith('.json'));
    for (const file of files) {
      fs.unlinkSync(path.join(RESULTS_DIR, file));
    }
  }

  generateMarkdown(): string {
    const lines: string[] = [];
    const timestamp = new Date().toISOString();
    lines.push('# E2E Validation Report');
    lines.push('');
    lines.push(`**Generated:** ${timestamp}`);
    lines.push(`**Total Tests:** ${this.results.length}`);
    const passed = this.results.filter(r => r.passed).length;
    const failed = this.results.filter(r => !r.passed).length;
    lines.push(`**Passed:** ${passed} | **Failed:** ${failed}`);
    lines.push('');

    // Group by suite
    const suites = [...new Set(this.results.map(r => r.suite))];
    lines.push('## Results by Suite');
    lines.push('');
    lines.push('| Suite | Test | Status | Duration | Error |');
    lines.push('|-------|------|--------|----------|-------|');
    for (const suite of suites) {
      const suiteResults = this.results.filter(r => r.suite === suite);
      for (const r of suiteResults) {
        const status = r.passed ? 'PASS' : 'FAIL';
        const dur = r.durationMs ? `${Math.round(r.durationMs / 1000)}s` : '-';
        const err = r.error ? r.error.substring(0, 80) : '-';
        lines.push(`| ${suite} | ${r.testName} | ${status} | ${dur} | ${err} |`);
      }
    }

    lines.push('');
    lines.push('## Summary by Suite');
    lines.push('');
    lines.push('| Suite | Passed | Failed | Total |');
    lines.push('|-------|--------|--------|-------|');
    for (const suite of suites) {
      const sr = this.results.filter(r => r.suite === suite);
      const sp = sr.filter(r => r.passed).length;
      const sf = sr.filter(r => !r.passed).length;
      lines.push(`| ${suite} | ${sp} | ${sf} | ${sr.length} |`);
    }

    if (failed > 0) {
      lines.push('');
      lines.push('## Failures');
      lines.push('');
      for (const r of this.results.filter(r => !r.passed)) {
        lines.push(`- **${r.suite}** / ${r.testName}: ${r.error || 'Unknown'}`);
      }
    }

    lines.push('');
    lines.push('---');
    lines.push(`*Report generated by E2EValidationReporter at ${timestamp}*`);
    return lines.join('\n');
  }

  writeReport(outputPath?: string): string {
    const filePath = outputPath || path.resolve(__dirname, '..', '..', '..', '..', 'test-results', 'E2E-VALIDATION-REPORT.md');
    const dir = path.dirname(filePath);
    if (!fs.existsSync(dir)) fs.mkdirSync(dir, { recursive: true });
    fs.writeFileSync(filePath, this.generateMarkdown(), 'utf-8');
    console.log(`[validation-reporter] Report written to: ${filePath}`);
    return filePath;
  }

  private appendResultToDisk(result: E2ETestResult): void {
    if (!this.persistenceTag) return;
    const filePath = path.join(RESULTS_DIR, `${this.persistenceTag}.json`);
    let existing: E2ETestResult[] = [];
    if (fs.existsSync(filePath)) {
      try { existing = JSON.parse(fs.readFileSync(filePath, 'utf-8')); } catch {}
    }
    existing.push(result);
    fs.writeFileSync(filePath, JSON.stringify(existing, null, 2), 'utf-8');
  }
}

export const e2eReporter = new E2EValidationReporter();
```

```typescript
export async function e2eValidationSetup(): Promise<void> {
  // Clean stale results
  E2EValidationReporter.cleanPersistedResults();

  const ctx = await request.newContext();

  // Check file-simulator health
  console.log(`[e2e-validation] Checking file-simulator at ${SIMULATOR_BASE}...`);

  if (!simHealth || !simHealth.ok()) {
    throw new Error(`FATAL: File-simulator unavailable at ${SIMULATOR_BASE}`);
  }
  console.log('[e2e-validation] File-simulator: HEALTHY');

  // Check EZ API health
  console.log(`[e2e-validation] Checking EZ API at ${EZ_API_BASE}...`);
  const ezHealth = await ctx.get(`${EZ_API_BASE}/health`).catch(() => null);

    throw new Error(`FATAL: EZ API unavailable at ${EZ_API_BASE}`);
  }
  console.log('[e2e-validation] EZ API: HEALTHY');

  await ctx.dispose();

  // Seed environment via DemoDataGenerator
```
