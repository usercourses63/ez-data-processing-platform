<!-- Docs URL logic (from src/Frontend/src/utils/docs-url.ts) -->
export const getDocsBaseUrl = (): string => {
  if (isDevelopment()) {
    // Dev: same hostname (minikube IP or localhost) with NodePort 30800
    return `http://${window.location.hostname}:30800`;
  }
  return '/docs';  // Production: path-based routing
};

// HelpPage.tsx redirects to:
const docsUrl = `${getDocsBaseUrl()}/docs/user-guide/`;
window.location.href = docsUrl;
