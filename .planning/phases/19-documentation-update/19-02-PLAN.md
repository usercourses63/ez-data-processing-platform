**Rewrite the environment detection and URL logic in `docs-url.ts`:**

Replace the `isProduction` function and `devPatterns` array with a proper RFC1918 private network detector:

```typescript
/**
 * Check if hostname is in RFC1918 private IP range or localhost
 */
const isPrivateNetwork = (hostname: string): boolean => {
  if (hostname === 'localhost' || hostname === '127.0.0.1') return true;
  if (hostname.startsWith('192.168.')) return true;
  if (hostname.startsWith('10.')) return true;
  // 172.16.0.0 - 172.31.255.255
  if (/^172\.(1[6-9]|2[0-9]|3[01])\./.test(hostname)) return true;
  return false;
};
```

Update `isProduction` to use the new function:
```typescript
export const isProduction = (): boolean => {
  return !isPrivateNetwork(window.location.hostname);
};
```
