  Success: boolean;
  ErrorMessage?: string;
  IsExportReachable: boolean;
  IsPvcBound: boolean;
  LatencyMs?: number;
}

export const testNasDeviceConnection = async (deviceId: string): Promise&lt;NasTestResult&gt; => {
  const response = await fetch(`${API_BASE_URL}/api/v1/nasdevices/${deviceId}/test-connection`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' }
  });
  if (!response.ok) {
    throw new Error(`Failed to test NAS connection: ${response.statusText}`);
  }
  return response.json();
};
