  StorageCapacity?: string;
  AccessMode?: string;
  ReclaimPolicy?: string;

}

export interface UpdateNasDeviceRequest extends Partial<CreateNasDeviceRequest> {}

export interface ProvisionNasDeviceRequest {
  Namespace?: string;
  CreatePv?: boolean;
  CreatePvc?: boolean;

  Success: boolean;
  PvCreated: boolean;
  PvcCreated: boolean;
  PvcBound: boolean;

Follow ServerListTab.tsx pattern.

```typescript
const NasDevicesTab: React.FC = () => {
  const { t } = useTranslation();
  const queryClient = useQueryClient();
  const [isModalVisible, setIsModalVisible] = useState(false);
  const [editingDevice, setEditingDevice] = useState<NasDevice | null>(null);
  const [provisioningDevice, setProvisioningDevice] = useState<string | null>(null);

  // Fetch NAS devices
  const { data: devices = [], isLoading, error } = useQuery({
    queryKey: nasDeviceQueryKeys.list(),
    queryFn: () => getNasDevices(),
  });

  // Mutations for delete, provision, test
  const deleteMutation = useMutation({ ... });
  const provisionMutation = useMutation({
    mutationFn: (id: string) => provisionNasDevice(id),
    onSuccess: (result) => {
      if (result.Success) {
        message.success(t('admin.nas.provisionSuccess'));
      } else {
        message.error(result.ErrorMessage || t('admin.nas.provisionError'));
      }
      queryClient.invalidateQueries({ queryKey: nasDeviceQueryKeys.all });
    },
  });
  const testConnectionMutation = useMutation({ ... });

  // Handlers
  const handleProvision = (device: NasDevice) => {
    Modal.confirm({
      title: t('admin.nas.confirmProvision'),
      content: t('admin.nas.provisionDescription', { name: device.Name }),
      onOk: () => provisionMutation.mutate(device.ID),
    });
  };

  return (
    <div>
      <Space style={{ marginBottom: 16, justifyContent: 'space-between', width: '100%' }}>
        <Typography.Title level={5}>{t('admin.nas.title')}</Typography.Title>
        <Button type="primary" icon={<PlusOutlined />} onClick={() => setIsModalVisible(true)}>
          {t('admin.nas.addDevice')}
        </Button>
      </Space>

      <NasDeviceTable
        devices={devices}
        loading={isLoading}
        onEdit={handleEdit}
        onDelete={handleDelete}
        onProvision={handleProvision}
        onTestConnection={handleTestConnection}
      />

      <NasDeviceModal
        visible={isModalVisible}
        device={editingDevice}
        onClose={() => { setIsModalVisible(false); setEditingDevice(null); }}
        onSuccess={() => { queryClient.invalidateQueries({ queryKey: nasDeviceQueryKeys.all }); setIsModalVisible(false); }}
      />
    </div>
  );
};
```
